cmake_minimum_required(VERSION 3.22)
if(NOT PROJECT_NAME)
    project(StutterLabCoreTests LANGUAGES CXX)
    enable_testing()
endif()
set(CMAKE_CXX_STANDARD 17)
add_executable(StutterLabCoreTests CoreTests.cpp)
find_package(Threads REQUIRED)
target_link_libraries(StutterLabCoreTests PRIVATE Threads::Threads)
if(MSVC)
    target_compile_options(StutterLabCoreTests PRIVATE /W4 /permissive-)
else()
    target_compile_options(StutterLabCoreTests PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME core COMMAND StutterLabCoreTests)
